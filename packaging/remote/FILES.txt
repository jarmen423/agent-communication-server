# Files a remote teammate needs
#
# This list is the minimal closure a remote machine must have on disk to
# run `python3 remote_agent_adapter.py --backend {shell|kilo|opencode}`
# against a hub. Verified by importing the adapter in a clean interpreter
# and recording every module actually pulled in (see note at the bottom).
#
# Everything is a flat file at the repo root or under worker_backends/.
# The install.sh helper copies exactly these files into the target
# directory; you can also copy them by hand or via sparse checkout.

# ── Entry point ──────────────────────────────────────────────────────
remote_agent_adapter.py

# ── Shared Python runtime (imported by the adapter) ─────────────────
nats_connect.py
worker_runtime.py
worker_events.py

# ── Backend package ──────────────────────────────────────────────────
# __init__.py is imported at adapter startup (re-exports HeadlessCliBackend
# and the *_spec factories used by the kilo / opencode backends).
worker_backends/__init__.py
worker_backends/headless_cli.py        # HeadlessCliBackend + HeadlessCliSpec
worker_backends/proc.py                # subprocess plumbing (imported by headless_cli.py)
worker_backends/presets.py             # kilo_spec(), opencode_spec(), hermes_spec(), agy_spec(), grok_spec()
worker_backends/sdk_agent.py           # SdkAgentBackend (imported by worker_backends/__init__.py)

# ── Pip requirements (copied so the target dir is self-contained) ───
requirements-remote.txt

# ── Optional docs (copy if you want the runbook on the target) ──────
# docs/JOIN_HUB.md
# docs/REMOTE_INSTALL.md

# ────────────────────────────────────────────────────────────────────
# NOT required for the shell/kilo/opencode backends:
#
#   worker_backends/acp_agent.py      — ACP-over-stdio (not wired into adapter)
#   worker_backends/acp_http.py       — ACP-over-HTTP; needs httpx
#   worker_backends/kilo_acp.py       — ACP variant of kilo
#   worker_backends/hermes_acp.py     — ACP variant of hermes
#   worker_backends/opencode_acp.py   — ACP variant of opencode
#   worker_backends/grok_acp.py       — ACP variant of grok
#   worker_backends/model_catalog.py  — only used by hub-server Rust side / local workers
#
# If you start using an ACP-over-HTTP backend remotely, add those files
# AND add `httpx>=0.27` to requirements-remote.txt.
# ────────────────────────────────────────────────────────────────────
